cmake_minimum_required(VERSION 3.10)
project(revev_engine LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_executable(revev_listening_test listening_mix_test.cpp)
add_executable(revev_limiter_test output_limiter_test.cpp)
add_executable(revev_compressor_test loudness_compressor_test.cpp)
add_executable(revev_turbo_test turbo_test.cpp)
add_executable(revev_mixer_test audio_mixer_test.cpp)
add_executable(revev_transmission_test transmission_model_test.cpp)
add_executable(revev_drive_model_test drive_rpm_model_test.cpp)
target_compile_options(revev_drive_model_test PRIVATE -O2)
add_executable(revev_tire_squeal_test tire_squeal_test.cpp)
target_compile_options(revev_limiter_test PRIVATE -O2)
target_compile_options(revev_listening_test PRIVATE -O2)
target_compile_options(revev_turbo_test PRIVATE -O2)
target_compile_options(revev_mixer_test PRIVATE -O2)
target_compile_options(revev_transmission_test PRIVATE -O2)
target_compile_options(revev_tire_squeal_test PRIVATE -O2)
set(ES ${CMAKE_CURRENT_SOURCE_DIR}/vendor/engine-sim)
file(GLOB ES_SOURCES ${ES}/src/*.cpp)
file(GLOB SCS_SOURCES ${ES}/dependencies/submodules/simple-2d-constraint-solver/src/*.cpp)
set(PIRANHA ${CMAKE_CURRENT_SOURCE_DIR}/vendor/piranha)
file(GLOB PIRANHA_SOURCES ${PIRANHA}/src/*.cpp ${PIRANHA}/generated/*.cpp)
file(GLOB SCRIPT_SOURCES ${ES}/scripting/src/*.cpp)
add_library(revev_core STATIC ${ES_SOURCES} ${SCS_SOURCES} ${PIRANHA_SOURCES} ${SCRIPT_SOURCES}
  engine_preset.cpp script_preset.cpp engine_runtime.cpp)
target_include_directories(revev_core PUBLIC ${PIRANHA}/include ${PIRANHA}/generated ${CMAKE_CURRENT_SOURCE_DIR}/vendor)
target_include_directories(revev_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${ES}/include)
# Upstream relied on transitive Windows headers for these standard definitions.
target_compile_options(revev_core PUBLIC -include ${CMAKE_CURRENT_SOURCE_DIR}/portable.h)
target_compile_options(revev_core PRIVATE -O2)
add_executable(revev_preset_smoke preset_smoke.cpp)
target_link_libraries(revev_preset_smoke PRIVATE revev_core)
add_executable(revev_synth_float_test synth_float_test.cpp)
target_link_libraries(revev_synth_float_test PRIVATE revev_core)
if(ANDROID)
  target_link_libraries(revev_core PUBLIC log)
  add_library(revev_audio SHARED android_audio.cpp)
  target_link_libraries(revev_audio PRIVATE revev_core aaudio android log)
  set_target_properties(revev_audio PROPERTIES LINK_FLAGS "-Wl,-z,max-page-size=16384")
  add_executable(revev_smoke smoke.cpp)
  target_link_libraries(revev_smoke PRIVATE revev_core)
else()
  find_package(Threads REQUIRED)
  add_executable(revev_smoke smoke.cpp)
  target_link_libraries(revev_smoke PRIVATE revev_core Threads::Threads)
endif()

add_executable(revev_prescribed_rpm_test prescribed_rpm_test.cpp)
target_link_libraries(revev_prescribed_rpm_test PRIVATE revev_core)

add_executable(revev_audio_bench audio_pipeline_bench.cpp)
target_link_libraries(revev_audio_bench PRIVATE revev_core)

add_executable(revev_output_continuity_test output_continuity_test.cpp)
